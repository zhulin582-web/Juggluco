cmake_minimum_required(VERSION 3.22)
project(juggluco_clarity LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(JUGGLUCO_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(clarity_core STATIC bytes.cpp cipher.cpp logging.cpp provider.cpp protocol.cpp records.cpp mapped.cpp account_store.cpp outbox.cpp client.cpp auth.cpp redirect.cpp http_error.cpp certificate.cpp certificate_crypto.cpp)
target_include_directories(clarity_core PUBLIC "${JUGGLUCO_CPP}" PRIVATE "${JUGGLUCO_CPP}/share")
target_compile_features(clarity_core PUBLIC cxx_std_20)
target_link_libraries(clarity_core PUBLIC z ${CMAKE_DL_LIBS})
set(CLARITY_PROVIDER_LIBRARY "libg7clarity.so" CACHE STRING "Packaged Clarity provider library name")
# Propagate the name to every translation unit using Engine/Provider defaults.
target_compile_definitions(clarity_core PUBLIC CLARITY_PROVIDER_LIBRARY="${CLARITY_PROVIDER_LIBRARY}")
if(ANDROID_ABI MATCHES "^(arm64-v8a|armeabi-v7a)$" OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|armv7.*|arm)$")
    set(CLARITY_NATIVE_ARM TRUE)
    target_sources(clarity_core PRIVATE engine.cpp)
else()
    if(ANDROID)
        message(FATAL_ERROR "The native Clarity provider requires arm64-v8a or armeabi-v7a")
    endif()
    target_sources(clarity_core PRIVATE engine_unavailable.cpp)
endif()
if(ANDROID)
    target_include_directories(clarity_core PRIVATE "${JUGGLUCO_CPP}/include")
    target_link_libraries(clarity_core PUBLIC log)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(clarity_core PUBLIC Threads::Threads m)
    enable_testing()
    if(CLARITY_NATIVE_ARM)
        add_executable(clarity_engine_test tests/engine_test.cpp)
        target_link_libraries(clarity_engine_test PRIVATE clarity_core)
        add_executable(clarity_compact_test tests/compact_test.cpp)
        target_link_libraries(clarity_compact_test PRIVATE clarity_core)
        # Optional offline replay against a locally supplied, private G7 response.
        add_executable(clarity_response_test tests/response_test.cpp)
        target_link_libraries(clarity_response_test PRIVATE clarity_core)
        add_executable(clarity_session_test tests/session_test.cpp)
        target_link_libraries(clarity_session_test PRIVATE clarity_core)
        add_executable(clarity_certificate_test tests/certificate_test.cpp)
        target_link_libraries(clarity_certificate_test PRIVATE clarity_core)
    endif()
    add_executable(clarity_cipher_test tests/cipher_test.cpp)
    target_link_libraries(clarity_cipher_test PRIVATE clarity_core)
    add_test(NAME clarity_cipher COMMAND clarity_cipher_test)
    if(CLARITY_BORINGSSL_CRYPTO_DIR)
        add_test(NAME clarity_cipher_boringssl COMMAND "${CMAKE_COMMAND}" -E env
            "LD_LIBRARY_PATH=${CLARITY_BORINGSSL_CRYPTO_DIR}:$ENV{LD_LIBRARY_PATH}"
            "$<TARGET_FILE:clarity_cipher_test>" BoringSSL)
    endif()
    add_executable(clarity_core_test tests/core_test.cpp tests/mapped_logging.cpp)
    target_link_libraries(clarity_core_test PRIVATE clarity_core)
    add_test(NAME clarity_core COMMAND clarity_core_test)
    add_executable(clarity_indexed_test tests/indexed_test.cpp tests/mapped_logging.cpp)
    target_link_libraries(clarity_indexed_test PRIVATE clarity_core)
    add_test(NAME clarity_indexed COMMAND clarity_indexed_test)
    add_executable(clarity_mapped_test tests/mapped_test.cpp tests/mapped_logging.cpp)
    target_link_libraries(clarity_mapped_test PRIVATE clarity_core)
    target_link_options(clarity_mapped_test PRIVATE "-Wl,--wrap=msync")
    add_test(NAME clarity_mapped COMMAND clarity_mapped_test)
    add_executable(clarity_event_test tests/event_test.cpp tests/mapped_logging.cpp)
    target_link_libraries(clarity_event_test PRIVATE clarity_core)
    add_test(NAME clarity_events COMMAND clarity_event_test)
    add_executable(clarity_backfill_test tests/backfill_test.cpp)
    target_link_libraries(clarity_backfill_test PRIVATE clarity_core)
    add_test(NAME clarity_backfill COMMAND clarity_backfill_test)
    add_executable(clarity_auth_test tests/auth_test.cpp)
    target_link_libraries(clarity_auth_test PRIVATE clarity_core)
    add_test(NAME clarity_auth COMMAND clarity_auth_test)
    add_executable(clarity_http_error_test tests/http_error_test.cpp)
    target_link_libraries(clarity_http_error_test PRIVATE clarity_core)
    add_test(NAME clarity_http_error COMMAND clarity_http_error_test)
    add_executable(clarity_redirect_test tests/redirect_test.cpp)
    target_link_libraries(clarity_redirect_test PRIVATE clarity_core)
    add_test(NAME clarity_redirect COMMAND clarity_redirect_test)
    find_package(OpenSSL QUIET)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    find_program(OPENSSL_TEST_EXECUTABLE openssl)
    if(Python3_FOUND AND OPENSSL_TEST_EXECUTABLE)
        add_executable(clarity_certificate_crypto_test tests/certificate_crypto_test.cpp)
        target_link_libraries(clarity_certificate_crypto_test PRIVATE clarity_core)
        add_test(NAME clarity_certificate_crypto COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/certificate_crypto_fixture.py"
            "$<TARGET_FILE:clarity_certificate_crypto_test>" "${OPENSSL_TEST_EXECUTABLE}" OpenSSL)
        set_tests_properties(clarity_certificate_crypto PROPERTIES TIMEOUT 30)
        set(CLARITY_BORINGSSL_CRYPTO_DIR "" CACHE PATH "Optional BoringSSL libcrypto.so directory for runtime regression tests")
        if(CLARITY_BORINGSSL_CRYPTO_DIR)
            add_test(NAME clarity_certificate_boringssl COMMAND "${CMAKE_COMMAND}" -E env
                "LD_LIBRARY_PATH=${CLARITY_BORINGSSL_CRYPTO_DIR}:$ENV{LD_LIBRARY_PATH}"
                "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/certificate_crypto_fixture.py"
                "$<TARGET_FILE:clarity_certificate_crypto_test>" "${OPENSSL_TEST_EXECUTABLE}" BoringSSL)
            set_tests_properties(clarity_certificate_boringssl PROPERTIES TIMEOUT 30)
        endif()
    endif()
    if(OpenSSL_FOUND AND Python3_FOUND AND OPENSSL_TEST_EXECUTABLE)
        add_executable(clarity_https_test tests/https_logging_test.cpp ../ICE/ContextHTTPS.cpp)
        target_include_directories(clarity_https_test PRIVATE "${JUGGLUCO_CPP}" "${JUGGLUCO_CPP}/share")
        target_link_libraries(clarity_https_test PRIVATE clarity_core OpenSSL::SSL OpenSSL::Crypto)
        add_test(NAME clarity_https_logging COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/https_fixture.py"
            "$<TARGET_FILE:clarity_https_test>" "${OPENSSL_TEST_EXECUTABLE}")
        set_tests_properties(clarity_https_logging PROPERTIES TIMEOUT 40)
    endif()
endif()
